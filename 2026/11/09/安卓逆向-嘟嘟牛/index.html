public void sendMobileCode(String str) {
    this.DEFAULT_TYPE = new TypeToken<RequestResult>() { // from class: com.dodonew.online.ui.LoginActivity.2
    }.getType();
    this.para.clear();      //将map清空
    this.para.put(IntentKey.PHONE, str); //put方法 第一个参数存入键 第二个存入该函数传递过来的参数值大概率手机号明文 
    requestNetwork("user/findPwd/sendMobileCode", this.para, this.DEFAULT_TYPE); // 盯着第二个参数即可
}

private void requestNetwork(final String str, Map<String, String> map, Type type) {
    showProgress();
    this.request = new JsonRequest(this, "http://api.dodovip.com/api/" + str, "", new Response.Listener<RequestResult>() { // from class: com.dodonew.online.ui.LoginActivity.7
        /* JADX WARN: Multi-variable type inference failed */
        @Override // com.android.volley.Response.Listener
        public void onResponse(RequestResult requestResult) {
            Log.e("登录信息", "login:=" + requestResult.response);
            if (requestResult.code.equals("1")) {
                if (str.equals(Config.URL_LOGIN) || str.equals(Config.URL_LOGIN_SMS)) {
                    DodonewOnlineApplication.setLoginUser((User) requestResult.data);
                    DodonewOnlineApplication.loginLabel = "mobile";
                    Utils.saveJson(LoginActivity.this, DodonewOnlineApplication.loginLabel, Config.LOGINLABEL_JSON);
                    User user = (User) new Gson().fromJson(requestResult.response, User.class);
                    LoginActivity.this.intentMainActivity();
                    EventBusManager.getInstace().getEventBus().post(new LoginSuccessInfo(user));
                    Log.e("登录注册", "userId:=" + DodonewOnlineApplication.getLoginUser().getUserId());
                } else if (str.equals("user/findPwd/sendMobileCode")) {
                    LoginActivity.this.timeCountUtil.start();
                    Log.e("LoginActivy", "sendCode:=" + requestResult.response);
                    LoginActivity.this.showToast("验证码已发送,请注意查收");
                }
            } else {
                LoginActivity.this.showToast(requestResult.message);
                Log.e("LoginActivy", "res:=" + requestResult.data);
            }
            LoginActivity.this.dissProgress();
        }
    }, this, type);
    this.request.addRequestMap(map, 0); //只在这里使用了 直接跟进去 
    DodonewOnlineApplication.addRequest(this.request, this);
}


public void addRequestMap(Map<String, String> map, int i) {
    String str = System.currentTimeMillis() + ""; //获取时间戳
    if (map == null) {
        map = new HashMap<>();
    }
    map.put(d.c.a.b, str); //放入map
    String strEncodeDesMap = RequestUtil.encodeDesMap(RequestUtil.paraMap(map, Config.BASE_APPEND, "sign"), this.desKey, this.desIV); //密文在这里生成的 看起来是des算法 但是是否魔改还需要进去细看 第一个参数需要搞清楚入参是什么
    JSONObject jSONObject = new JSONObject();
    try {
        jSONObject.put("Encrypt", strEncodeDesMap); //很明显创建一个json 在向上找strEncodeDesMap即可
        this.mRequestBody = jSONObject + "";
    } catch (JSONException e) {
        e.printStackTrace();
    }
}


public static String paraMap(Map<String, String> map, String str, String str2) {
    try {
        Set<String> setKeySet = map.keySet(); // 取出map的键放入setKeySet
        StringBuilder sb = new StringBuilder(); // 创建可变字符串，用来拼最终待签名串。
        ArrayList arrayList = new ArrayList(); // 创建数组
        for (String str3 : setKeySet) {
            arrayList.add(str3 + "=" + map.get(str3));
        }// 将map转化为 键=值 的格式 并放入数组中
        Collections.sort(arrayList); //对 arrayList 做自然排序。因为元素是 String，所以按字符串字典序排
        for (int i = 0; i < arrayList.size(); i++) {
            sb.append((String) arrayList.get(i));
            sb.append(a.b);// &
        }//将arrayList里的内容按&分割拼接
        sb.append("key=" + str);//在拼接固定盐值
        map.put(str2, Utils.md5(sb.toString()).toUpperCase()); // 进行MD5并放入原map中 键为sign
        String json = new GsonBuilder().serializeNulls().create().toJson(sortMapByKey(map));// 把 map 序列化成 JSON 字符串 得到"{'sign':'a10ssscccacsc','phone':'19999999999'}
        Log.w(AppConfig.DEBUG_TAG, json + "   result");
        return json; //将结果return回去
    } catch (Exception e) {
        e.printStackTrace();
        return "";
    }
}
